cmake_minimum_required(VERSION 3.20)
project(xylonic-fiber LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(PkgConfig QUIET)
find_package(Threads REQUIRED)
option(XF_BUILD_TESTS "Build the test executables" ON)
if(ANDROID)
  # Android: SDL2 comes from the sources fetched by android/setup.sh (add_subdirectory happens in android/app/jni).
  if(NOT TARGET SDL2::SDL2)
    message(FATAL_ERROR "Build through android/app/jni/CMakeLists.txt, which provides the SDL2 target")
  endif()
  add_library(PkgConfig_SDL2 INTERFACE)
  target_link_libraries(PkgConfig_SDL2 INTERFACE SDL2::SDL2)
  add_library(PkgConfig::SDL2 ALIAS PkgConfig_SDL2)
  set(XF_BUILD_TESTS OFF)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  pkg_check_modules(SDL2_IMAGE REQUIRED IMPORTED_TARGET SDL2_image)
endif()
include(FetchContent)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
FetchContent_Declare(mbedtls GIT_REPOSITORY https://github.com/Mbed-TLS/mbedtls.git GIT_TAG v3.6.7 GIT_SHALLOW TRUE GIT_SUBMODULES_RECURSE TRUE)
FetchContent_MakeAvailable(mbedtls)
add_library(net STATIC src/net/url.cpp src/net/stream.cpp src/net/http.cpp src/net/discovery.cpp src/net/remote.cpp)
target_compile_options(net PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(net PUBLIC mbedtls mbedx509 mbedcrypto yyjson Threads::Threads)
add_library(yyjson STATIC third_party/yyjson.c)
target_compile_options(yyjson PRIVATE -O2)
add_library(cachecore STATIC src/cache/metadata.cpp src/cache/journal.cpp)
target_compile_options(cachecore PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
add_library(subsonic STATIC src/subsonic/client.cpp)
target_compile_options(subsonic PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(subsonic PUBLIC net yyjson)
set_source_files_properties(src/ui/svg.cpp PROPERTIES COMPILE_OPTIONS "-w")
add_library(ui STATIC src/ui/gfx.cpp src/ui/ui.cpp src/ui/themes.cpp src/ui/svg.cpp)
target_compile_options(ui PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(ui PUBLIC PkgConfig::SDL2)
if(NOT ANDROID)
  pkg_check_modules(SYSTEMD IMPORTED_TARGET libsystemd)
endif()
add_library(appcore STATIC src/app/app.cpp src/app/app_views.cpp src/app/shell.cpp src/app/library.cpp src/app/pages.cpp src/app/search.cpp src/app/panel.cpp src/app/nowplaying.cpp src/app/settings.cpp src/app/login.cpp src/app/playlists.cpp src/app/dialogs.cpp src/app/carmode.cpp src/app/mini.cpp src/app/remote_ui.cpp src/app/queue.cpp src/app/worker.cpp src/app/images.cpp src/app/config.cpp src/app/secrets.cpp src/app/mpris.cpp src/app/model_json.cpp src/app/store.cpp src/app/downloads.cpp)
target_compile_options(appcore PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(appcore PUBLIC subsonic ui audio cachecore PkgConfig::SDL2 Threads::Threads ${CMAKE_DL_LIBS})
if(NOT ANDROID)
  target_link_libraries(appcore PUBLIC PkgConfig::SDL2_IMAGE)
endif()
add_library(audio STATIC src/audio/decoder.cpp src/audio/player.cpp src/audio/source.cpp src/audio/dsp.cpp)
target_compile_options(audio PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(audio PUBLIC PkgConfig::SDL2 Threads::Threads net)
if(ANDROID)
  add_library(xylonic-fiber SHARED src/main.cpp)  # loaded by SDLActivity as libmain.so (renamed in android/app/jni)
  set_target_properties(xylonic-fiber PROPERTIES OUTPUT_NAME main)
else()
  add_executable(xylonic-fiber src/main.cpp)
  add_custom_command(TARGET xylonic-fiber POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets ${CMAKE_BINARY_DIR}/assets)
endif()
target_compile_options(xylonic-fiber PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(xylonic-fiber PRIVATE appcore)
if(SYSTEMD_FOUND AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_definitions(appcore PUBLIC XF_MPRIS)
  target_link_libraries(appcore PUBLIC PkgConfig::SYSTEMD)
endif()

if(XF_BUILD_TESTS)
enable_testing()
add_test(NAME smoke COMMAND ${CMAKE_COMMAND} -E env SDL_VIDEODRIVER=dummy SDL_AUDIODRIVER=dummy $<TARGET_FILE:xylonic-fiber> --selftest)

find_program(FFMPEG ffmpeg)
if(FFMPEG)
  add_executable(audio_test tests/audio_test.cpp)
  target_compile_options(audio_test PRIVATE -Wall -Wextra)
  target_link_libraries(audio_test PRIVATE audio)
  add_test(NAME gen_media COMMAND ${FFMPEG} -loglevel error -y -f lavfi -i sine=frequency=440:duration=30 -b:a 192k ${CMAKE_BINARY_DIR}/test.mp3)
  set_tests_properties(gen_media PROPERTIES FIXTURES_SETUP media)
  add_test(NAME gen_short COMMAND ${FFMPEG} -loglevel error -y -f lavfi -i sine=frequency=440:duration=2 -b:a 128k ${CMAKE_BINARY_DIR}/test_short.mp3)
  set_tests_properties(gen_short PROPERTIES FIXTURES_SETUP media)
  add_test(NAME audio COMMAND ${CMAKE_COMMAND} -E env SDL_AUDIODRIVER=dummy $<TARGET_FILE:audio_test> ${CMAKE_BINARY_DIR}/test.mp3 ${CMAKE_BINARY_DIR}/test_short.mp3)
  set_tests_properties(audio PROPERTIES FIXTURES_REQUIRED media)
endif()

add_executable(subsonic_test tests/subsonic_test.cpp)
target_compile_options(subsonic_test PRIVATE -Wall -Wextra)
target_link_libraries(subsonic_test PRIVATE subsonic)
add_test(NAME subsonic COMMAND subsonic_test)
add_executable(net_test tests/net_test.cpp)
target_compile_options(net_test PRIVATE -Wall -Wextra)
target_link_libraries(net_test PRIVATE net Threads::Threads)
add_test(NAME net COMMAND net_test)
find_program(OPENSSL openssl)
if(OPENSSL)
  add_test(NAME tls COMMAND ${CMAKE_SOURCE_DIR}/tests/tls_test.sh $<TARGET_FILE:net_test>)
endif()

add_executable(ui_test tests/ui_test.cpp)
target_compile_options(ui_test PRIVATE -Wall -Wextra)
target_link_libraries(ui_test PRIVATE ui)
add_test(NAME ui COMMAND ${CMAKE_COMMAND} -E env SDL_VIDEODRIVER=dummy $<TARGET_FILE:ui_test>)
set_tests_properties(ui PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})

add_executable(queue_test tests/queue_test.cpp)
target_compile_options(queue_test PRIVATE -Wall -Wextra)
target_link_libraries(queue_test PRIVATE appcore)
add_test(NAME queue COMMAND queue_test)

add_executable(secrets_test tests/secrets_test.cpp)
target_link_libraries(secrets_test PRIVATE appcore)
add_test(NAME secrets COMMAND secrets_test)

add_executable(downloads_test tests/downloads_test.cpp)
target_link_libraries(downloads_test PRIVATE appcore)
add_test(NAME downloads COMMAND downloads_test)

add_executable(themes_test tests/themes_test.cpp)
target_link_libraries(themes_test PRIVATE ui)
add_test(NAME themes COMMAND themes_test)

add_executable(discovery_test tests/discovery_test.cpp)
target_link_libraries(discovery_test PRIVATE net Threads::Threads)
add_test(NAME discovery COMMAND discovery_test)

add_executable(dsp_test tests/dsp_test.cpp)
target_link_libraries(dsp_test PRIVATE audio)
add_test(NAME dsp COMMAND dsp_test)


add_executable(mpris_test tests/mpris_test.cpp)
target_link_libraries(mpris_test PRIVATE appcore)
add_test(NAME mpris COMMAND mpris_test)

add_executable(remote_test tests/remote_test.cpp)
target_link_libraries(remote_test PRIVATE net Threads::Threads)
add_test(NAME remote COMMAND remote_test)
add_executable(cache_test tests/cache_test.cpp)
target_compile_options(cache_test PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(cache_test PRIVATE cachecore)
add_test(NAME cache COMMAND cache_test)
add_executable(journal_test tests/journal_test.cpp)
target_compile_options(journal_test PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(journal_test PRIVATE cachecore)
add_test(NAME journal COMMAND journal_test)
endif()
