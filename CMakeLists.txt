cmake_minimum_required(VERSION 3.20)
project(xylonic-fiber LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
include(FetchContent)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
FetchContent_Declare(mbedtls GIT_REPOSITORY https://github.com/Mbed-TLS/mbedtls.git GIT_TAG v3.6.7 GIT_SHALLOW TRUE GIT_SUBMODULES_RECURSE TRUE)
FetchContent_MakeAvailable(mbedtls)
add_library(net STATIC src/net/url.cpp src/net/stream.cpp src/net/http.cpp)
target_compile_options(net PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(net PUBLIC mbedtls mbedx509 mbedcrypto)
add_library(yyjson STATIC third_party/yyjson.c)
target_compile_options(yyjson PRIVATE -O2)
add_library(subsonic STATIC src/subsonic/client.cpp)
target_compile_options(subsonic PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(subsonic PUBLIC net yyjson)
add_library(ui STATIC src/ui/gfx.cpp)
target_compile_options(ui PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(ui PUBLIC PkgConfig::SDL2)
add_library(audio STATIC src/audio/decoder.cpp src/audio/player.cpp src/audio/source.cpp)
target_compile_options(audio PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(audio PUBLIC PkgConfig::SDL2 Threads::Threads net)
add_executable(xylonic-fiber src/main.cpp)
target_compile_options(xylonic-fiber PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(xylonic-fiber PRIVATE PkgConfig::SDL2)
enable_testing()
add_test(NAME smoke COMMAND ${CMAKE_COMMAND} -E env SDL_VIDEODRIVER=dummy SDL_AUDIODRIVER=dummy $<TARGET_FILE:xylonic-fiber> --selftest)

find_program(FFMPEG ffmpeg)
if(FFMPEG)
  add_executable(audio_test tests/audio_test.cpp)
  target_compile_options(audio_test PRIVATE -Wall -Wextra)
  target_link_libraries(audio_test PRIVATE audio)
  add_test(NAME gen_media COMMAND ${FFMPEG} -loglevel error -y -f lavfi -i sine=frequency=440:duration=30 -b:a 192k ${CMAKE_BINARY_DIR}/test.mp3)
  set_tests_properties(gen_media PROPERTIES FIXTURES_SETUP media)
  add_test(NAME audio COMMAND ${CMAKE_COMMAND} -E env SDL_AUDIODRIVER=dummy $<TARGET_FILE:audio_test> ${CMAKE_BINARY_DIR}/test.mp3)
  set_tests_properties(audio PROPERTIES FIXTURES_REQUIRED media)
endif()

add_executable(subsonic_test tests/subsonic_test.cpp)
target_compile_options(subsonic_test PRIVATE -Wall -Wextra)
target_link_libraries(subsonic_test PRIVATE subsonic)
add_test(NAME subsonic COMMAND subsonic_test)
add_executable(net_test tests/net_test.cpp)
target_compile_options(net_test PRIVATE -Wall -Wextra)
target_link_libraries(net_test PRIVATE net Threads::Threads)
add_test(NAME net COMMAND net_test)
find_program(OPENSSL openssl)
if(OPENSSL)
  add_test(NAME tls COMMAND ${CMAKE_SOURCE_DIR}/tests/tls_test.sh $<TARGET_FILE:net_test>)
endif()

add_executable(ui_test tests/ui_test.cpp)
target_compile_options(ui_test PRIVATE -Wall -Wextra)
target_link_libraries(ui_test PRIVATE ui)
add_test(NAME ui COMMAND ${CMAKE_COMMAND} -E env SDL_VIDEODRIVER=dummy $<TARGET_FILE:ui_test>)
set_tests_properties(ui PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
