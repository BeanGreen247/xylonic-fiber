cmake_minimum_required(VERSION 3.20)
project(xylonic-fiber LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(XF_LTO "Link-time optimisation and section GC for Release builds (binary 3.73 -> 2.91 MB, same CPU: docs/baseline.md)" ON)
if(XF_LTO AND NOT ANDROID AND NOT IOS AND CMAKE_BUILD_TYPE STREQUAL "Release")
  include(CheckIPOSupported)
  check_ipo_supported(RESULT xf_ipo_ok OUTPUT xf_ipo_msg)
  if(xf_ipo_ok)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
    add_compile_options(-ffunction-sections -fdata-sections)
    if(APPLE)
      add_link_options(-Wl,-dead_strip)
    else()
      add_link_options(-Wl,--gc-sections)
    endif()
  endif()
endif()
find_package(PkgConfig QUIET)
find_package(Threads REQUIRED)
option(XF_BUILD_TESTS "Build the test executables" ON)
option(XF_APP_STORE "App store build: shows the thank-you banner in Settings, hides nothing else" OFF)
if(ANDROID)
  # Android: SDL2 comes from the sources fetched by android/setup.sh (add_subdirectory happens in android/app/jni).
  if(NOT TARGET SDL2::SDL2)
    message(FATAL_ERROR "Build through android/app/jni/CMakeLists.txt, which provides the SDL2 target")
  endif()
  add_library(PkgConfig_SDL2 INTERFACE)
  target_link_libraries(PkgConfig_SDL2 INTERFACE SDL2::SDL2)
  add_library(PkgConfig::SDL2 ALIAS PkgConfig_SDL2)
  set(XF_BUILD_TESTS OFF)
elseif(IOS)
  # iOS: build SDL2 from source as a static library (cmake -G Xcode -DCMAKE_SYSTEM_NAME=iOS, see docs/APPLE.md). No SDL_image:
  # artwork is decoded with stb_image, as on Android.
  include(FetchContent)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(sdl2 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-2.30.9 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(sdl2)
  add_library(PkgConfig_SDL2 INTERFACE)
  target_link_libraries(PkgConfig_SDL2 INTERFACE SDL2::SDL2-static)
  add_library(PkgConfig::SDL2 ALIAS PkgConfig_SDL2)
  set(XF_BUILD_TESTS OFF)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  pkg_check_modules(SDL2_IMAGE REQUIRED IMPORTED_TARGET SDL2_image)
endif()
include(FetchContent)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_USER_CONFIG_FILE ${CMAKE_CURRENT_LIST_DIR}/third_party/mbedtls_user_config.h CACHE FILEPATH "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
FetchContent_Declare(mbedtls GIT_REPOSITORY https://github.com/Mbed-TLS/mbedtls.git GIT_TAG v3.6.7 GIT_SHALLOW TRUE GIT_SUBMODULES_RECURSE TRUE)
FetchContent_MakeAvailable(mbedtls)
add_library(net STATIC src/net/url.cpp src/net/stream.cpp src/net/http.cpp src/net/discovery.cpp src/net/remote.cpp)
target_compile_options(net PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(net PUBLIC mbedtls mbedx509 mbedcrypto yyjson Threads::Threads)
add_library(yyjson STATIC third_party/yyjson.c)
target_compile_options(yyjson PRIVATE -O2)
add_library(cachecore STATIC src/cache/metadata.cpp src/cache/journal.cpp src/cache/generation.cpp src/cache/media.cpp)
target_compile_options(cachecore PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
add_library(subsonic STATIC src/subsonic/client.cpp)
target_compile_options(subsonic PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(subsonic PUBLIC net yyjson)
set_source_files_properties(src/ui/svg.cpp PROPERTIES COMPILE_OPTIONS "-w")
add_library(ui STATIC src/ui/gfx.cpp src/ui/ui.cpp src/ui/themes.cpp src/ui/svg.cpp)
target_compile_options(ui PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(ui PUBLIC PkgConfig::SDL2)
if(NOT ANDROID AND NOT APPLE)
  pkg_check_modules(SYSTEMD IMPORTED_TARGET libsystemd)
endif()
add_library(appcore STATIC src/app/app.cpp src/app/app_views.cpp src/app/shell.cpp src/app/library.cpp src/app/pages.cpp src/app/search.cpp src/app/panel.cpp src/app/nowplaying.cpp src/app/settings.cpp src/app/login.cpp src/app/playlists.cpp src/app/dialogs.cpp src/app/carmode.cpp src/app/mini.cpp src/app/remote_ui.cpp src/app/queue.cpp src/app/worker.cpp src/app/images.cpp src/app/config.cpp src/app/secrets.cpp src/app/mpris.cpp src/app/model_json.cpp src/app/store.cpp src/app/downloads.cpp src/app/metered.cpp src/app/precache.cpp src/app/library_actions.cpp src/app/log.cpp src/app/downloads_page.cpp src/app/storage.cpp src/app/storage_ui.cpp src/app/modals.cpp src/app/android_bridge.cpp)
target_compile_options(appcore PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
if(XF_APP_STORE)
  target_compile_definitions(appcore PUBLIC XF_APP_STORE)
endif()
target_link_libraries(appcore PUBLIC subsonic ui audio cachecore PkgConfig::SDL2 Threads::Threads ${CMAKE_DL_LIBS})
if(NOT ANDROID AND NOT IOS)
  target_link_libraries(appcore PUBLIC PkgConfig::SDL2_IMAGE)
endif()
if(APPLE)
  # Keychain (credentials) and CoreFoundation; iOS also reads the cellular reachability flag for "auto-offline on mobile data".
  target_link_libraries(appcore PUBLIC "-framework Security" "-framework CoreFoundation")
  if(IOS)
    target_link_libraries(appcore PUBLIC "-framework SystemConfiguration")
  endif()
endif()
add_library(audio STATIC src/audio/decoder.cpp src/audio/player.cpp src/audio/source.cpp src/audio/dsp.cpp)
target_compile_options(audio PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(audio PUBLIC PkgConfig::SDL2 Threads::Threads net)
if(ANDROID)
  add_library(xylonic-fiber SHARED src/main.cpp)  # loaded by SDLActivity as libmain.so (renamed in android/app/jni)
  set_target_properties(xylonic-fiber PROPERTIES OUTPUT_NAME main)
elseif(APPLE)
  # .app bundle (macOS and iOS). SDL_GetBasePath() is Contents/Resources on macOS and the bundle root on iOS; assets go there.
  add_executable(xylonic-fiber MACOSX_BUNDLE src/main.cpp)
  if(IOS)
    set(XF_PLIST ${CMAKE_SOURCE_DIR}/cmake/Info-ios.plist.in)
  else()
    set(XF_PLIST ${CMAKE_SOURCE_DIR}/cmake/Info-macos.plist.in)
  endif()
  set_target_properties(xylonic-fiber PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST ${XF_PLIST}
    MACOSX_BUNDLE_GUI_IDENTIFIER org.xylonic.fiber
    MACOSX_BUNDLE_BUNDLE_NAME "Xylonic Fiber"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "0.1.0"
    MACOSX_BUNDLE_BUNDLE_VERSION "1"
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER org.xylonic.fiber)
  if(IOS)
    set_target_properties(xylonic-fiber PROPERTIES XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2" XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET "13.0")
    set(XF_ASSET_DEST $<TARGET_BUNDLE_DIR:xylonic-fiber>/assets)
  else()
    set(XF_ASSET_DEST $<TARGET_BUNDLE_CONTENT_DIR:xylonic-fiber>/Resources/assets)
  endif()
  add_custom_command(TARGET xylonic-fiber POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets ${XF_ASSET_DEST})
else()
  add_executable(xylonic-fiber src/main.cpp)
  add_custom_command(TARGET xylonic-fiber POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets ${CMAKE_BINARY_DIR}/assets)
endif()
target_compile_options(xylonic-fiber PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(xylonic-fiber PRIVATE appcore)
if(SYSTEMD_FOUND AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_definitions(appcore PUBLIC XF_MPRIS)
  target_link_libraries(appcore PUBLIC PkgConfig::SYSTEMD)
endif()
